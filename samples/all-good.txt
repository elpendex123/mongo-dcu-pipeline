db.properties.find({ "property_type": "condo" }, { "property_id": 1, "listing_price": 1, "_id": 0 })

// Aggregations, within the DocumentDB-supported stage set.

db.properties.aggregate([{ "$match": { "listing_status": "sold" } }, { "$project": { "property_id": 1, "listing_price": 1, "_id": 0 } }, { "$limit": 10 }])
