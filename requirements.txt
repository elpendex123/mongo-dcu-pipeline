# Runtime dependencies.
#
# Pinned exactly: the image built in CI and the local virtual environment must
# resolve the same versions, or a query that validates locally could behave
# differently in QA.

# MongoDB driver. Also provides bson, which the parser uses to turn
# extended-JSON forms into real ObjectId, datetime and Decimal128 values.
pymongo==4.18.1

# Tolerant JSON parser. Query lines are written by hand in mongosh syntax, so
# unquoted keys, single quotes and trailing commas are normal rather than
# errors.
json5==0.15.0

# AWS SDK: S3 for file movement, SES for run summaries.
boto3==1.43.92

# MySQL driver. Pure Python, so the image needs no compiler and no MySQL
# client libraries.
PyMySQL==1.2.0

# Metrics endpoint scraped by Prometheus.
prometheus-client==0.26.0
